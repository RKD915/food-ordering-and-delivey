const app = express();
app.use(cors());

  .then(() => console.log('Successfully connected to MongoDB!'))
  .catch((err) => console.error('MongoDB connection error:', err));

// Test Route
app.get('/', (req, res) => {
  res.send('Food Delivery API is running...');
});

// Authentication Routes
app.post('/api/register', async (req, res) => {
  try {
    const { name, email, password } = req.body;
    const existingUser = await User.findOne({ email });
    if (existingUser) return res.status(400).json({ message: 'User already exists' });

    const salt = await bcrypt.genSalt(10);
    const hashedPassword = await bcrypt.hash(password, salt);

    const newUser = new User({ name, email, password: hashedPassword });
    await newUser.save();

    res.status(201).json({ message: 'User registered successfully' });
  } catch (error) {
    res.status(500).json({ message: 'Server error' });
  }
});

app.post('/api/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    const user = await User.findOne({ email });
    if (!user) return res.status(404).json({ message: 'User not found' });

    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) return res.status(400).json({ message: 'Invalid credentials' });

    const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: '1h' });

    res.json({ token, user: { id: user._id, name: user.name, email: user.email } });
  } catch (error) {
    res.status(500).json({ message: 'Server error' });
  }
});

// Food Routes
app.get('/api/foods', async (req, res) => {
  try {
    const foods = await Food.find();
    res.json(foods);
  } catch (error) {
    res.status(500).json({ message: 'Server error' });
  }
});

// Order Routes
app.post('/api/orders', async (req, res) => {
  try {
    const { items, totalAmount, paymentMethod } = req.body;
    // Assuming simple mock user for now, or require a logged in user payload
    const mockUserId = new mongoose.Types.ObjectId(); 

    const newOrder = new Order({
      user: mockUserId, // Usually from JWT auth token
      items: items.map(i => ({ food: i._id, quantity: i.quantity })),

    res.status(201).json({ message: 'Order created', orderId: savedOrder._id });
  } catch (error) {
    res.status(500).json({ message: 'Server error' });
  }

import { useState, useEffect } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { Minus, Plus, Trash2 } from 'lucide-react';
import styles from './Cart.module.css';

const Cart = () => {
  const navigate = useNavigate();
  const [cart, setCart] = useState([]);
  const [paymentMethod, setPaymentMethod] = useState('');
  const [verificationAmount, setVerificationAmount] = useState('');

  useEffect(() => {
    const savedCart = JSON.parse(localStorage.getItem('cart') || '[]');
    setCart(savedCart);
  }, []);

  const updateQuantity = (id, delta) => {
    const newCart = cart.map(item => {
      if (item._id === id) {
        return { ...item, quantity: Math.max(1, item.quantity + delta) };
      }
      return item;
    });
    setCart(newCart);
    localStorage.setItem('cart', JSON.stringify(newCart));
  };

  const removeItem = (id) => {
    const newCart = cart.filter(item => item._id !== id);
    setCart(newCart);
    localStorage.setItem('cart', JSON.stringify(newCart));
  };

  const subtotal = cart.reduce((acc, item) => acc + (item.price * item.quantity), 0);
  const deliveryFee = cart.length > 0 ? 50 : 0;
  const total = subtotal + deliveryFee;

  const handleCheckout = async () => {
    if (!paymentMethod) {
      alert("Please select a payment method before checking out!");
      return;
    }
    if (parseFloat(verificationAmount) !== total) {
      alert(`Payment verification failed! Please enter exactly ₹${total} to approve the transaction.`);
      return;
    }
    if (cart.length === 0) {
      alert("Your cart is empty!");
      return;
    }
    
    try {
      const response = await fetch('http://localhost:5000/api/orders', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          items: cart,
          totalAmount: total,
          paymentMethod
        })
      });

      if (response.ok) {
        localStorage.removeItem('cart');
        navigate('/order-confirmed');
      } else {
        alert("Checkout failed. Try again.");
      }
    } catch (err) {
      alert("Error confirming order.");
    }
  };

  return (
    <div className={`container ${styles.cartPage}`}>
      <div className={styles.cartLayout}>
        {/* Left Side: Cart Items */}
        <div className={styles.cartItems}>
          {cart.length === 0 ? (
            <div className={`card ${styles.emptyCart}`}>
              <h2>Your cart is empty</h2>
              <Link to="/" className="btn btn-primary" style={{marginTop: '1rem', display: 'inline-block'}}>Browse Menu</Link>
            </div>
          ) : (
            cart.map(item => (
              <div key={item._id} className={`card ${styles.cartItem}`}>
                <img 
                  src={`http://localhost:5173${item.imageUrl}`} 
                  alt={item.name} 
                  className={styles.itemImage}
                />
                
                <div className={styles.itemDetails}>
                  <div className={styles.itemHeader}>
                    <h3 className={styles.itemTitle}>{item.name}</h3>
                    <button className={styles.deleteBtn} onClick={() => removeItem(item._id)}>
                      <Trash2 size={18} />
                    </button>
                  </div>
                  <p className={styles.itemDesc}>{item.description}</p>
                  
                  <div className={styles.itemFooter}>
                    <span className={styles.itemPrice}>₹{item.price * item.quantity}</span>
                    <div className={styles.quantityControl}>
                      <button className={styles.qtyBtn} onClick={() => updateQuantity(item._id, -1)}><Minus size={14} /></button>
                      <span className={styles.qty}>{item.quantity}</span>
                      <button className={styles.qtyBtn} onClick={() => updateQuantity(item._id, 1)}><Plus size={14} /></button>
                    </div>
                  </div>
                </div>
              </div>
            ))
          )}
        </div>

        {/* Right Side: Order Summary & Payments */}
        <div className={styles.orderSummaryWrapper}>
          <div className={`card ${styles.orderSummary}`}>
            <h2 className={styles.summaryTitle}>Order Summary</h2>
            
            <div className={styles.summaryRow}>
              <span className={styles.summaryLabel}>Subtotal</span>
              <span className={styles.summaryValue}>₹{subtotal}</span>
            </div>
            <div className={styles.summaryRow}>
              <span className={styles.summaryLabel}>Delivery Fee</span>
              <span className={styles.summaryValue}>₹{deliveryFee}</span>
            </div>
            
            <hr className={styles.divider} />
            
            <div className={`${styles.summaryRow} ${styles.totalRow}`}>
              <span className={styles.summaryLabel}>Total</span>
              <span className={styles.totalValue}>₹{total}</span>
            </div>

            <div style={{ marginTop: '2rem', marginBottom: '1rem' }}>
              <h3 style={{ fontSize: '1rem', marginBottom: '1rem' }}>Select Payment Method</h3>
              <div style={{ display: 'flex', flexDirection: 'column', gap: '0.5rem' }}>
                <label style={{ display: 'flex', alignItems: 'center', gap: '0.5rem', cursor: 'pointer' }}>
                  <input type="radio" name="payment" onChange={() => setPaymentMethod('UPI')} />
                  UPI (PhonePe, BHIM)
                </label>
                <label style={{ display: 'flex', alignItems: 'center', gap: '0.5rem', cursor: 'pointer' }}>
                  <input type="radio" name="payment" onChange={() => setPaymentMethod('Google Pay')} />
                  Google Pay
                </label>
                <label style={{ display: 'flex', alignItems: 'center', gap: '0.5rem', cursor: 'pointer' }}>
                  <input type="radio" name="payment" onChange={() => setPaymentMethod('Paytm')} />
                  Paytm
                </label>
              </div>
            </div>

            {paymentMethod && (
              <div style={{ padding: '1rem', background: 'rgba(252,76,2,0.1)', borderRadius: '8px', marginBottom: '1rem', border: '1px solid var(--primary)' }}>
                <h4 style={{ marginBottom: '0.5rem', color: 'var(--primary)' }}>{paymentMethod} Verification</h4>
                <p style={{ fontSize: '0.9rem', marginBottom: '0.5rem' }}>Please verify by typing the exact total amount (₹{total}):</p>
                <input 
                  type="number" 
                  value={verificationAmount} 
                  onChange={(e) => setVerificationAmount(e.target.value)} 
                  placeholder={`₹${total}`}
                  style={{ width: '100%', padding: '0.8rem', borderRadius: '4px', border: '1px solid #ccc' }}
                />
                {verificationAmount && parseFloat(verificationAmount) !== total && (
                  <p style={{ color: 'red', fontSize: '0.8rem', marginTop: '0.5rem' }}>Amount doesn't match total!</p>
                )}
              </div>
            )}
            
            <button 
              className={`btn btn-primary ${styles.checkoutBtn}`}
              onClick={handleCheckout}
              disabled={cart.length === 0 || parseFloat(verificationAmount) !== total}
              style={{ opacity: cart.length === 0 || parseFloat(verificationAmount) !== total ? 0.5 : 1 }}
            >
              Pay ₹{total} & Checkout
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};

export default Cart;
